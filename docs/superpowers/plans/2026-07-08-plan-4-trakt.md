```swift
import Foundation

/// Trakt.tv API v2 client. MainActor because tokens mutate and the app is
/// MainActor-driven; all I/O is async URLSession.
@MainActor
public final class TraktClient {
    public var tokens: TraktTokens?
    /// Fired whenever a 401-triggered refresh succeeds; the app persists them.
    public var onTokensRefreshed: ((TraktTokens) -> Void)?

    let clientID: String
    private let clientSecret: String
    private let session: URLSession
    private let base = URL(string: "https://api.trakt.tv")!

    static let decoder: JSONDecoder = {
        let decoder = JSONDecoder()
        let fractional = ISO8601DateFormatter()
        fractional.formatOptions = [.withInternetDateTime, .withFractionalSeconds]
        let plain = ISO8601DateFormatter()
        decoder.dateDecodingStrategy = .custom { d in
            let raw = try d.singleValueContainer().decode(String.self)
            if let date = fractional.date(from: raw) ?? plain.date(from: raw) { return date }
            throw DecodingError.dataCorrupted(.init(codingPath: d.codingPath,
                                                    debugDescription: "Bad date: \(raw)"))
        }
        return decoder
    }()

    public init(clientID: String, clientSecret: String, tokens: TraktTokens? = nil,
                session: URLSession = .shared) {
        self.clientID = clientID
        self.clientSecret = clientSecret
        self.tokens = tokens
        self.session = session
    }

    // MARK: Device auth

    public func requestDeviceCode() async throws -> TraktDeviceCode {
        let data = try await post(path: "/oauth/device/code",
                                  body: ["client_id": clientID],
                                  allowedStatus: [200])
        return try Self.decoder.decode(TraktDeviceCode.self, from: data)
    }

    /// One poll step. nil = still pending (caller sleeps `interval` and retries).
    public func pollDeviceToken(deviceCode: String) async throws -> TraktTokens? {
        let (data, status) = try await postRaw(path: "/oauth/device/token",
                                               body: ["code": deviceCode,
                                                      "client_id": clientID,
                                                      "client_secret": clientSecret])
        switch status {
        case 200:
            let fetched = try Self.decoder.decode(TraktTokens.self, from: data)
            tokens = fetched
            return fetched
        case 400, 429: return nil                       // pending / slow down
        case 404, 409, 410, 418: throw TraktError.denied
        default: throw TraktError.http(status)
        }
    }

    @discardableResult
    public func refreshTokens() async throws -> TraktTokens {
        guard let tokens else { throw TraktError.unauthorized }
        let (data, status) = try await postRaw(path: "/oauth/token",
                                               body: ["refresh_token": tokens.refreshToken,
                                                      "client_id": clientID,
                                                      "client_secret": clientSecret,
                                                      "redirect_uri": "urn:ietf:wg:oauth:2.0:oob",
                                                      "grant_type": "refresh_token"])
        guard status == 200 else { throw TraktError.unauthorized }
        let fetched = try Self.decoder.decode(TraktTokens.self, from: data)
        self.tokens = fetched
        onTokensRefreshed?(fetched)
        return fetched
    }

    // MARK: Request infra

    private func request(path: String, method: String, body: Data?,
                         authorized: Bool) -> URLRequest {
        var req = URLRequest(url: base.appending(path: path))
        req.httpMethod = method
        req.httpBody = body
        req.setValue("application/json", forHTTPHeaderField: "Content-Type")
        req.setValue("2", forHTTPHeaderField: "trakt-api-version")
        req.setValue(clientID, forHTTPHeaderField: "trakt-api-key")
        if authorized, let tokens {
            req.setValue("Bearer \(tokens.accessToken)", forHTTPHeaderField: "Authorization")
        }
        return req
    }

    private func postRaw(path: String, body: [String: String]) async throws -> (Data, Int) {
        let payload = try JSONEncoder().encode(body)
        let req = request(path: path, method: "POST", body: payload, authorized: false)
        let (data, response) = try await session.data(for: req)
        guard let http = response as? HTTPURLResponse else { throw TraktError.badResponse }
        return (data, http.statusCode)
    }

    private func post(path: String, body: [String: String],
                      allowedStatus: Set<Int>) async throws -> Data {
        let (data, status) = try await postRaw(path: path, body: body)
        guard allowedStatus.contains(status) else { throw TraktError.http(status) }
        return data
    }

    /// Authorized call with a single automatic token refresh on 401.
    func send(path: String, method: String = "GET", body: Data? = nil) async throws -> Data {
        guard tokens != nil else { throw TraktError.unauthorized }
        let (data, response) = try await session.data(
            for: request(path: path, method: method, body: body, authorized: true))
        guard let http = response as? HTTPURLResponse else { throw TraktError.badResponse }
        switch http.statusCode {
        case 200, 201, 204: return data
        case 401:
            try await refreshTokens()   // throws .unauthorized on failure
            let (data2, response2) = try await session.data(
                for: request(path: path, method: method, body: body, authorized: true))
            guard let http2 = response2 as? HTTPURLResponse,
                  (200...299).contains(http2.statusCode) else {
                throw TraktError.unauthorized
            }
            return data2
        default: throw TraktError.http(http.statusCode)
        }
    }
}
```

```swift
    /// URLProtocol exposes bodies as streams; read whichever form is set.
    var bodyData: Data {
        if let httpBody { return httpBody }
        guard let stream = httpBodyStream else { return Data() }
        stream.open(); defer { stream.close() }
        var data = Data()
        let size = 4096
        let buffer = UnsafeMutablePointer<UInt8>.allocate(capacity: size)
        defer { buffer.deallocate() }
        while stream.hasBytesAvailable {
            let read = stream.read(buffer, maxLength: size)
            if read <= 0 { break }
            data.append(buffer, count: read)
        }
        return data
    }
```

```swift
// MARK: Payload encoding

struct TraktIDsPayload: Encodable { let tmdb: Int }
struct TraktRefPayload: Encodable { let ids: TraktIDsPayload }
struct TraktEpisodePayload: Encodable { let season: Int; let number: Int }

struct ScrobbleBody: Encodable {
    var movie: TraktRefPayload?
    var show: TraktRefPayload?
    var episode: TraktEpisodePayload?
    let progress: Double
}

struct RatingsBody: Encodable {
    struct Item: Encodable { let rating: Int; let ids: TraktIDsPayload }
    var movies: [Item]?
    var shows: [Item]?
}

extension TraktClient {
    // MARK: Scrobbling

    public func scrobble(_ action: TraktScrobbleAction, media: TraktMedia,
                         progress: Double) async throws {
        var body = ScrobbleBody(progress: progress)
        switch media {
        case .movie(let id), .show(let id):
            body.movie = TraktRefPayload(ids: TraktIDsPayload(tmdb: id))
        case .episode(let showID, let season, let episode):
            body.show = TraktRefPayload(ids: TraktIDsPayload(tmdb: showID))
            body.episode = TraktEpisodePayload(season: season, number: episode)
        }
        _ = try await send(path: "/scrobble/\(action.rawValue)", method: "POST",
                           body: try JSONEncoder().encode(body))
    }

    // MARK: Sync pulls

    public func playback() async throws -> [TraktPlaybackEntry] {
        try Self.decoder.decode([TraktPlaybackEntry].self,
                                from: try await send(path: "/sync/playback"))
    }

    public func watchedMovies() async throws -> [TraktWatchedMovie] {
        try Self.decoder.decode([TraktWatchedMovie].self,
                                from: try await send(path: "/sync/watched/movies"))
    }

    public func watchedShows() async throws -> [TraktWatchedShow] {
        try Self.decoder.decode([TraktWatchedShow].self,
                                from: try await send(path: "/sync/watched/shows"))
    }

    // MARK: Ratings

    /// Episodes rate their show (spec: episode-level ratings out of v1).
    public func rate(media: TraktMedia, rating: Int) async throws {
        var body = RatingsBody()
        let clamped = min(max(rating, 1), 10)
        switch media {
        case .movie(let id):
            body.movies = [.init(rating: clamped, ids: TraktIDsPayload(tmdb: id))]
        case .show(let id), .episode(let id, _, _):
            body.shows = [.init(rating: clamped, ids: TraktIDsPayload(tmdb: id))]
        }
        _ = try await send(path: "/sync/ratings", method: "POST",
                           body: try JSONEncoder().encode(body))
    }

    public func ratedMovies() async throws -> [TraktRatingEntry] {
        try Self.decoder.decode([TraktRatingEntry].self,
                                from: try await send(path: "/sync/ratings/movies"))
    }

    public func ratedShows() async throws -> [TraktRatingEntry] {
        try Self.decoder.decode([TraktRatingEntry].self,
                                from: try await send(path: "/sync/ratings/shows"))
    }
}
```

```swift
    /// Remote (Trakt) playback position. Unlike saveProgress this keeps the
    /// caller's timestamp so newer-wins comparisons stay correct.
    public func applyRemoteProgress(itemKey: String, position: Double,
                                    duration: Double, watchedAt: Date) throws {
        let descriptor = FetchDescriptor<WatchState>(
            predicate: #Predicate { $0.itemKey == itemKey })
        if let state = try context.fetch(descriptor).first {
            state.positionSeconds = position
            state.durationSeconds = duration
            state.lastWatchedAt = watchedAt
            state.finished = false
        } else {
            let state = WatchState(itemKey: itemKey, positionSeconds: position,
                                   durationSeconds: duration, finished: false)
            state.lastWatchedAt = watchedAt
            context.insert(state)
        }
        try context.save()
    }

    /// Remote (Trakt) watched marker: finished, position/duration preserved.
    public func applyRemoteWatched(itemKey: String, watchedAt: Date) throws {
        let descriptor = FetchDescriptor<WatchState>(
            predicate: #Predicate { $0.itemKey == itemKey })
        if let state = try context.fetch(descriptor).first {
            state.finished = true
            state.lastWatchedAt = watchedAt
        } else {
            let state = WatchState(itemKey: itemKey, positionSeconds: 0,
                                   durationSeconds: 0, finished: true)
            state.lastWatchedAt = watchedAt
            context.insert(state)
        }
        try context.save()
    }
```

```swift
import Foundation

/// Pulls Trakt playback progress + watched history into the local store.
/// Newest-timestamp-wins in both directions; local state is never touched
/// when it is newer than Trakt's.
@MainActor
public struct TraktSyncer {
    public struct Summary: Equatable, Sendable {
        public var playbackApplied = 0
        public var watchedApplied = 0
        public init() {}
    }

    let client: TraktClient
    let store: LibraryStore

    public init(client: TraktClient, store: LibraryStore) {
        self.client = client
        self.store = store
    }

    public func pull() async throws -> Summary {
        let items = try store.items()
        var movieKeys: [Int: String] = [:]          // movie tmdb -> itemKey
        var episodeKeys: [String: String] = [:]     // "showTmdb#s#e" -> itemKey
        for item in items {
            guard let id = item.tmdbID else { continue }
            if item.mediaKind == "movie" {
                movieKeys[id] = item.itemKey
            } else if let s = item.season, let e = item.episode {
                episodeKeys["\(id)#\(s)#\(e)"] = item.itemKey
            }
        }
        var states = Dictionary(uniqueKeysWithValues:
            (try store.allWatchStates()).map { ($0.itemKey, ($0.lastWatchedAt, $0.durationSeconds, $0.finished)) })
        var summary = Summary()

        for entry in try await client.playback() {
            let key: String? = switch entry.type {
            case "movie": entry.movie?.ids.tmdb.flatMap { movieKeys[$0] }
            case "episode": entry.show?.ids.tmdb.flatMap { showID in
                entry.episode.flatMap { episodeKeys["\(showID)#\($0.season)#\($0.number)"] }
            }
            default: nil
            }
            guard let key, let pausedAt = entry.pausedAt else { continue }
            let local = states[key]
            // Percent -> seconds needs a locally known duration.
            guard let duration = local?.1, duration > 0,
                  pausedAt > (local?.0 ?? .distantPast) else { continue }
            try store.applyRemoteProgress(itemKey: key,
                                          position: entry.progress / 100 * duration,
                                          duration: duration, watchedAt: pausedAt)
            states[key] = (pausedAt, duration, false)
            summary.playbackApplied += 1
        }

        func markWatched(key: String, at watchedAt: Date) throws {
            let local = states[key]
            guard local?.2 != true, watchedAt > (local?.0 ?? .distantPast) else { return }
            try store.applyRemoteWatched(itemKey: key, watchedAt: watchedAt)
            states[key] = (watchedAt, local?.1 ?? 0, true)
            summary.watchedApplied += 1
        }

        for watched in try await client.watchedMovies() {
            guard let id = watched.movie.ids.tmdb, let key = movieKeys[id] else { continue }
            try markWatched(key: key, at: watched.lastWatchedAt)
        }
        for show in try await client.watchedShows() {
            guard let showID = show.show.ids.tmdb else { continue }
            for season in show.seasons {
                for episode in season.episodes {
                    guard let key = episodeKeys["\(showID)#\(season.number)#\(episode.number)"]
                    else { continue }
                    try markWatched(key: key, at: episode.lastWatchedAt)
                }
            }
        }
        return summary
    }
}
```

```swift
import Foundation
import Observation
import StreamCore

/// App-side Trakt facade: Keychain-backed credentials/tokens, one shared
/// client, best-effort wrappers that never throw into UI code.
@Observable
@MainActor
final class TraktService {
    static let shared = TraktService()

    private let keychain = KeychainStore()
    private(set) var clientID: String
    private(set) var clientSecret: String
    private(set) var tokens: TraktTokens?
    private(set) var needsReauth = false
    private var client: TraktClient?
    /// Session cache: "movie-603"/"show-1399" -> user rating.
    private var ratingsCache: [String: Int]?

    init() {
        clientID = keychain.get("trakt-client-id") ?? ""
        clientSecret = keychain.get("trakt-client-secret") ?? ""
        if let raw = keychain.get("trakt-tokens"),
           let decoded = try? JSONDecoder().decode(TraktTokens.self, from: Data(raw.utf8)) {
            tokens = decoded
        }
        rebuildClient()
    }

    var isConfigured: Bool { !clientID.isEmpty && !clientSecret.isEmpty }
    var isConnected: Bool { tokens != nil }

    private func rebuildClient() {
        guard isConfigured else { client = nil; return }
        let built = TraktClient(clientID: clientID, clientSecret: clientSecret, tokens: tokens)
        built.onTokensRefreshed = { [weak self] fresh in self?.persist(tokens: fresh) }
        client = built
    }

    private func persist(tokens: TraktTokens?) {
        self.tokens = tokens
        if let tokens, let data = try? JSONEncoder().encode(tokens),
           let raw = String(data: data, encoding: .utf8) {
            keychain.set(raw, account: "trakt-tokens")
        } else {
            keychain.delete(account: "trakt-tokens")
        }
    }

    func saveCredentials(clientID: String, clientSecret: String) {
        self.clientID = clientID.trimmingCharacters(in: .whitespacesAndNewlines)
        self.clientSecret = clientSecret.trimmingCharacters(in: .whitespacesAndNewlines)
        keychain.set(self.clientID, account: "trakt-client-id")
        keychain.set(self.clientSecret, account: "trakt-client-secret")
        rebuildClient()
    }

    func disconnect() {
        persist(tokens: nil)
        needsReauth = false
        ratingsCache = nil
        rebuildClient()
    }

    // MARK: Device flow

    func requestDeviceCode() async throws -> TraktDeviceCode {
        guard let client else { throw TraktError.unauthorized }
        return try await client.requestDeviceCode()
    }

    /// Polls until authorized, denied, or the code expires. True on success.
    func awaitAuthorization(code: TraktDeviceCode) async -> Bool {
        guard let client else { return false }
        let deadline = Date().addingTimeInterval(TimeInterval(code.expiresIn))
        while Date() < deadline, !Task.isCancelled {
            do {
                if let fetched = try await client.pollDeviceToken(deviceCode: code.deviceCode) {
                    persist(tokens: fetched)
                    needsReauth = false
                    return true
                }
            } catch {
                return false
            }
            try? await Task.sleep(for: .seconds(code.interval))
        }
        return false
    }

    // MARK: Best-effort wrappers

    func scrobble(_ action: TraktScrobbleAction, media: TraktMedia, progress: Double) {
        guard let client, isConnected else { return }
        Task {
            do { try await client.scrobble(action, media: media, progress: progress) }
            catch TraktError.unauthorized { needsReauth = true }
            catch { NSLog("Trakt scrobble failed: \(error)") }
        }
    }

    func pullSync(store: LibraryStore) async {
        guard let client, isConnected else { return }
        do {
            _ = try await TraktSyncer(client: client, store: store).pull()
        } catch TraktError.unauthorized {
            needsReauth = true
        } catch {
            NSLog("Trakt sync failed: \(error)")
        }
    }

    func rating(for media: TraktMedia) async -> Int? {
        guard let client, isConnected else { return nil }
        if ratingsCache == nil {
            var cache: [String: Int] = [:]
            if let movies = try? await client.ratedMovies() {
                for entry in movies {
                    if let id = entry.movie?.ids.tmdb { cache["movie-\(id)"] = entry.rating }
                }
            }
            if let shows = try? await client.ratedShows() {
                for entry in shows {
                    if let id = entry.show?.ids.tmdb { cache["show-\(id)"] = entry.rating }
                }
            }
            ratingsCache = cache
        }
        return ratingsCache?[cacheKey(media)]
    }

    func rate(media: TraktMedia, rating: Int) {
        guard let client, isConnected else { return }
        ratingsCache?[cacheKey(media)] = rating
        Task {
            do { try await client.rate(media: media, rating: rating) }
            catch TraktError.unauthorized { needsReauth = true }
            catch { NSLog("Trakt rate failed: \(error)") }
        }
    }

    private func cacheKey(_ media: TraktMedia) -> String {
        switch media {
        case .movie(let id): "movie-\(id)"
        case .show(let id), .episode(let id, _, _): "show-\(id)"
        }
    }
}

extension TraktMedia {
    /// nil when the item is unmatched (invisible to Trakt).
    init?(playItem: PlayItem) {
        guard let id = playItem.tmdbID else { return nil }
        if let season = playItem.season, let episode = playItem.episode {
            self = .episode(showTMDBID: id, season: season, episode: episode)
        } else {
            self = .movie(tmdbID: id)
        }
    }
}
```
